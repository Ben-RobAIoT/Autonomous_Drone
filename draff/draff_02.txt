# main.py
import time
import threading
import os
from flask import Flask, jsonify, request, send_from_directory, Response
from flask_cors import CORS

from config import SystemConfig, FlightConfig, PIDConfig
from core.connection import DroneConnection
from core.state_machine import FlightState
from sensors.monitor import SensorManager
from sensors.vision import VisionManager
from control.commander import FlightCommander
from missions.fire_round_2 import FireMission
from utils.logger import FlightLogger
from utils.safety import SafetyManager

app = Flask(__name__, static_folder='dashboard')
CORS(app)

# Biến toàn cục
drone_telemetry = {}
commander_ptr = None
vision_ptr = None
current_state = FlightState.IDLE # Đưa ra ngoài để Web có thể sửa đổi

@app.route('/')
def index():
    return send_from_directory(app.static_folder, 'index.html')

@app.route('/telemetry')
def get_telemetry():
    return jsonify(drone_telemetry)

@app.route('/command', methods=['POST'])
def send_command():
    global current_state
    action = request.json.get('action')
    if action == 'ARM':
        commander_ptr.arm()
        current_state = FlightState.TAKEOFF # Cập nhật trạng thái bay
    elif action == 'LAND':
        commander_ptr.land_safely()
        current_state = FlightState.RETURN_LAND
    elif action == 'DISARM':
        commander_ptr.disarm()
        current_state = FlightState.DISARMED
    return jsonify({"status": "received", "action": action})

def gen_frames():
    while True:
        if vision_ptr:
            frame = vision_ptr.get_encoded_frame()
            if frame:
                yield (b'--frame\r\n'
                       b'Content-Type: image/jpeg\r\n\r\n' + frame + b'\r\n')
        time.sleep(0.05)

@app.route('/video_feed')
def video_feed():
    return Response(gen_frames(), mimetype='multipart/x-mixed-replace; boundary=frame')

def run_web_server():
    app.run(host='0.0.0.0', port=5000, debug=False, use_reloader=False)

def main():
    global commander_ptr, vision_ptr, drone_telemetry, current_state
    
    drone = DroneConnection(SystemConfig.PORT, SystemConfig.BAUD)
    sensors = SensorManager(drone)
    commander = FlightCommander(drone)
    vision = VisionManager() 
    
    commander_ptr = commander
    vision_ptr = vision
    
    logger = FlightLogger()
    safety = SafetyManager(commander, sensors)
    fire_mission = FireMission(commander, vision)
    
    threading.Thread(target=run_web_server, daemon=True).start()
    print("[MAIN] Web Server đang chạy tại cổng 5000")
    
    time.sleep(2) 
    print("\n[READY] Hệ thống đã sẵn sàng. Vui lòng thao tác trên Web GCS!")
    
    try:
        while True:
            data = sensors.get_all()
            # Dùng .get() để tránh lỗi nếu key chưa kịp khởi tạo
            current_alt = data.get('alt', 0.0)
            
            drone_telemetry = {
                "state": current_state.name,
                "alt": round(current_alt, 2),
                "battery": round(data.get('battery_v', 0.0), 2),
                "vibration": round(data.get('vibration_z', 0.0), 2),
                "flow_qual": data.get('flow_qual', 0)
            }
            
            # Kiểm tra an toàn
            if current_state != FlightState.IDLE and not safety.check_all():
                current_state = FlightState.FAILSAFE
                
            # Nếu đang IDLE (chưa bấm nút trên web), chỉ cập nhật Telemetry rồi lặp lại
            if current_state == FlightState.IDLE:
                time.sleep(0.1)
                continue

            # --- LOGIC BAY KHI ĐÃ ARM ---
            error = FlightConfig.HOVER_TARGET_ALT - current_alt
            
            if abs(error) < 0.2 and current_state == FlightState.TAKEOFF:
                current_state = FlightState.MISSION_ACTIVE
                print("\n[MISSION] Đã đạt độ cao. Bắt đầu chữa cháy!")

            if current_state == FlightState.MISSION_ACTIVE:
                fire_mission.run_step(current_alt)

            pwm = PIDConfig.PWM_HOVER + (error * PIDConfig.ALT_KP)
            pwm = max(PIDConfig.PWM_MIN, min(PIDConfig.PWM_MAX, pwm))
            
            commander.set_rc_throttle(pwm)
            drone.send_heartbeat()
            logger.log(current_state.name, data)
            
            print(f"[{current_state.name}] Cao độ: {current_alt:.2f}m | PWM: {int(pwm)}", end='\r')
            time.sleep(1.0 / FlightConfig.LOOP_HZ)

    except KeyboardInterrupt:
        print("\n\n[STOP] Ngắt từ bàn phím. Đang hạ cánh...")
        commander.land_safely()
        current_state = FlightState.RETURN_LAND
        
        while sensors.get_altitude() > FlightConfig.LANDING_STOP_ALT:
            time.sleep(0.5)
        
        commander.disarm()
        current_state = FlightState.DISARMED
        
    finally:
        sensors.stop()
        vision.stop()
        logger.close()
        print("\n[FINISH] Hệ thống đã tắt an toàn.")

if __name__ == '__main__':
    main()
